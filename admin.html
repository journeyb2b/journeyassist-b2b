function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Fetch Packages var sheetPkg = ss.getSheetByName("Packages"); var dataPkg = sheetPkg ? sheetPkg.getDataRange().getValues() : []; var packages = []; if (dataPkg.length > 1) { var headers = dataPkg[0]; for (var i = 1; i < dataPkg.length; i++) { var row = dataPkg[i]; var pkg = {}; for (var j = 0; j < headers.length; j++) { var key = String(headers[j]).trim().toLowerCase().replace(/\s+/g, ''); pkg[key] = row[j]; } packages.push({ code: pkg.code || "JA-" + i, title: pkg.title || "Holiday Package", destination: String(pkg.destination || "other").trim().toLowerCase(), region: String(pkg.region || "international").trim().toLowerCase(), duration: pkg.duration || "3N/4D", category: pkg.category || "Standard", b2bPrice: pkg.b2bprice || "On Request", mrpPrice: pkg.mrpprice || "", inclusions: pkg.inclusions || "", status: String(pkg.status || "OPEN").toUpperCase(), travelDates: pkg.traveldates || "", ratesGrid: pkg.ratesgrid || "", itinerary: pkg.itinerary || "" }); } } // Fetch Agents (For Admin) var agents = []; if (e && e.parameter && e.parameter.admin === "true") { var sheetAgents = ss.getSheetByName("Agents"); var dataAgents = sheetAgents ? sheetAgents.getDataRange().getValues() : []; if (dataAgents.length > 1) { for (var k = 1; k < dataAgents.length; k++) { agents.push({ fullName: dataAgents[k][1], agencyName: dataAgents[k][2], phone: dataAgents[k][3], city: dataAgents[k][4], email: dataAgents[k][5] }); } } } return ContentService.createTextOutput(JSON.stringify({ success: true, packages: packages, agents: agents })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); // ACTION: BULK ADD PACKAGES FROM CSV if (data.action === "bulk_add_packages" && data.packages) { var sheetPkg = ss.getSheetByName("Packages") || ss.insertSheet("Packages"); if (sheetPkg.getLastRow() === 0) { sheetPkg.appendRow(["code", "title", "destination", "region", "duration", "category", "b2bPrice", "mrpPrice", "inclusions", "status", "travelDates", "status_active", "ratesGrid", "itinerary"]); } var rowsToAppend = []; for (var i = 0; i < data.packages.length; i++) { var p = data.packages[i]; rowsToAppend.push([ p.code || ("JA-" + (sheetPkg.getLastRow() + i + 1)), p.title || "", String(p.destination || "other").toLowerCase(), String(p.region || "international").toLowerCase(), p.duration || "", p.category || "", p.b2bPrice || "", p.mrpPrice || "", p.inclusions || "", p.status || "OPEN", p.travelDates || "", "Active", p.ratesGrid || "", p.itinerary || "" ]); } if (rowsToAppend.length > 0) { sheetPkg.getRange(sheetPkg.getLastRow() + 1, 1, rowsToAppend.length, rowsToAppend[0].length).setValues(rowsToAppend); } return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Bulk packages uploaded successfully!" })).setMimeType(ContentService.MimeType.JSON); } // ACTION: ADD SINGLE PACKAGE if (data.action === "add_package") { var sheetPkg = ss.getSheetByName("Packages") || ss.insertSheet("Packages"); if (sheetPkg.getLastRow() === 0) { sheetPkg.appendRow(["code", "title", "destination", "region", "duration", "category", "b2bPrice", "mrpPrice", "inclusions", "status", "travelDates", "status_active", "ratesGrid", "itinerary"]); } var pkgCode = "JA-" + (sheetPkg.getLastRow()); sheetPkg.appendRow([ pkgCode, data.title, data.destination, data.region, data.duration, data.category, data.b2bPrice, "", data.inclusions, "OPEN", data.travelDates, "Active", data.ratesGrid, data.itinerary ]); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Package added successfully!" })).setMimeType(ContentService.MimeType.JSON); } // ACTION: AGENT SIGNIN / SIGNUP / CUSTOM QUOTE if (data.action === "signin" || data.action === "signup") { var sheet = ss.getSheetByName("Agents") || ss.insertSheet("Agents"); if (sheet.getLastRow() === 0) { sheet.appendRow(["Timestamp", "FullName", "AgencyName", "Phone", "City", "Email", "Password"]); } if (data.action === "signup") { sheet.appendRow([new Date(), data.fullName, data.agencyName, data.phone, data.city, data.email, data.password]); MailApp.sendEmail("journeyassist11@gmail.com", "New Agent", data.fullName + " - " + data.agencyName); return ContentService.createTextOutput(JSON.stringify({ success: true, agent: { name: data.fullName, agency: data.agencyName, email: data.email, phone: data.phone } })).setMimeType(ContentService.MimeType.JSON); } if (data.action === "signin") { var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][5] === data.email && rows[i][6] === data.password) { return ContentService.createTextOutput(JSON.stringify({ success: true, agent: { name: rows[i][1], agency: rows[i][2], email: rows[i][5], phone: rows[i][3] } })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Invalid credentials!" })).setMimeType(ContentService.MimeType.JSON); } } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } }
