function openPackageDetails(index) { const p = allPackages[index]; if (!p) return; const modalContent = document.getElementById('modalPackageContent'); // 1. HOTEL DETAILS PARSE & RENDER let hotelsDisplay = ''; try { let hotelData = p.hotels; if (typeof hotelData === 'string' && hotelData.trim() !== '' && hotelData !== '[]') { hotelData = JSON.parse(hotelData); } if (Array.isArray(hotelData) && hotelData.length > 0) { let hotelCards = ''; hotelData.forEach(h => { hotelCards += `
${h.category || 'Category'} ${h.room || 'Room'}

${h.hotels || 'N/A'}

`; }); hotelsDisplay = `

Hotel Categories & Room Mapping

${hotelCards}
`; } } catch (e) { hotelsDisplay = ''; } // 2. ITINERARY LINKS PARSE & RENDER let linksDisplay = ''; try { let linksData = p.itineraryLink; if (typeof linksData === 'string' && linksData.trim() !== '' && linksData !== '[]') { try { linksData = JSON.parse(linksData); } catch(err) { linksData = [{ title: 'View Itinerary', url: p.itineraryLink }]; } } if (Array.isArray(linksData) && linksData.length > 0) { let linkButtons = ''; linksData.forEach(l => { if (l && l.url) { linkButtons += ` ${l.title || 'View Detailed Itinerary'} `; } }); if (linkButtons !== '') { linksDisplay = `

Official Itinerary Documents

${linkButtons}
`; } } } catch (e) { linksDisplay = ''; } // 3. RATE GRID PARSE & RENDER let rateGridHtml = ''; try { const grid = typeof p.ratesGrid === 'string' ? JSON.parse(p.ratesGrid) : p.ratesGrid; if (Array.isArray(grid) && grid.length > 0) { rateGridHtml = `
${grid.map(g => ` `).join('')}
Category Rate
${g.category || g.hotel || 'Option'} ${g.price || g.rate || '-'}
`; } else { rateGridHtml = `

Standard rates applicable as listed above.

`; } } catch (e) { rateGridHtml = `

Standard rates applicable.

`; } // 4. DAY-WISE ITINERARY PARSE & RENDER let itineraryDisplay = ''; if (p.itinerary && p.itinerary !== '[]' && p.itinerary !== '""' && p.itinerary.trim() !== '') { try { let parsed = typeof p.itinerary === 'string' ? JSON.parse(p.itinerary) : p.itinerary; if (Array.isArray(parsed) && parsed.length > 0) { let rows = ''; parsed.forEach((d, idx) => { if (d && ((d.title && d.title.trim() !== '') || (d.desc && d.desc.trim() !== ''))) { rows += `
${d.title || ('Day ' + (idx + 1))}

${d.desc || ''}

`; } }); if (rows !== '') { itineraryDisplay = `

Day-Wise Tour Itinerary Details

${rows}
`; } } } catch (e) { itineraryDisplay = ''; } } modalContent.innerHTML = `
${p.destination || 'Package'} (${p.region || 'General'})

${p.title}

${p.travelDates || ''}

Pricing & Inclusions

B2B Net Rate: ${p.b2bPrice}
Inclusions: ${p.inclusions}
${hotelsDisplay} ${linksDisplay}

Rate Grid / Options

${rateGridHtml}
${itineraryDisplay}
Open Dynamic Brochure
`; document.getElementById('packageModal').classList.remove('hidden'); }
