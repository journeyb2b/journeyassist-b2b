function doPost(e) { try { var data = JSON.parse(e.postData.contents); var action = data.action; var ss = SpreadsheetApp.getActiveSpreadsheet(); // ------------------------------------------------------------- // 1. ACTION: CUSTOM PACKAGE QUOTE REQUEST // ------------------------------------------------------------- if (action === "custom_quote") { var sheet = ss.getSheetByName("Custom_Queries") || ss.insertSheet("Custom_Queries"); // Auto-create Headers if sheet is empty if (sheet.getLastRow() === 0) { sheet.appendRow([ "Timestamp", "Agent Name", "Agency Name", "Agent Email", "Agent Phone", "Destination", "Travel Date", "Nights", "Adults", "Kids", "Category", "Requirements" ]); } var timestamp = new Date(); sheet.appendRow([ timestamp, data.agentName, data.agencyName, data.agentEmail, data.agentPhone, data.destination, data.travelDate, data.nights, data.adults, data.kids, data.category, data.requirements ]); // Email Notification to journeyassist11@gmail.com var myEmail = "journeyassist11@gmail.com"; var emailSubject = "🚨 New Custom Quote Request from " + data.agencyName; var emailBody = "Hi Journey Assist Team,\n\n" + "A new custom package query has been submitted by a B2B agent:\n\n" + "--------------------------------------------------\n" + "Agent Name: " + data.agentName + "\n" + "Agency Name: " + data.agencyName + "\n" + "Email: " + data.agentEmail + "\n" + "Phone: " + data.agentPhone + "\n" + "--------------------------------------------------\n" + "Destination: " + data.destination + "\n" + "Travel Date: " + data.travelDate + "\n" + "Duration: " + data.nights + "\n" + "Pax Details: " + data.adults + " Adults, " + data.kids + " Kids\n" + "Hotel Category: " + data.category + "\n" + "Special Requirements: " + data.requirements + "\n" + "--------------------------------------------------\n\n" + "Please process this query on priority."; MailApp.sendEmail(myEmail, emailSubject, emailBody); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Your custom quote request has been submitted successfully! Email alert sent to team." })).setMimeType(ContentService.MimeType.JSON); } // ------------------------------------------------------------- // 2. ACTION: AGENT SIGN UP / REGISTRATION // ------------------------------------------------------------- if (action === "signup") { var agentSheet = ss.getSheetByName("Agents") || ss.insertSheet("Agents"); if (agentSheet.getLastRow() === 0) { agentSheet.appendRow(["Timestamp", "Agent ID", "Full Name", "Agency Name", "Mobile", "City", "Email", "Password"]); } // Check if email already exists var agentData = agentSheet.getDataRange().getValues(); for (var i = 1; i < agentData.length; i++) { if (agentData[i][6] === data.email) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Email address is already registered. Please sign in." })).setMimeType(ContentService.MimeType.JSON); } } var agentId = "JA-AGT-" + Math.floor(1000 + Math.random() * 9000); var regTimestamp = new Date(); agentSheet.appendRow([ regTimestamp, agentId, data.fullName, data.agencyName, data.phone, data.city, data.email, data.password ]); // Registration Notification Email to journeyassist11@gmail.com var alertEmail = "journeyassist11@gmail.com"; var regSubject = "🎉 New Agent Registered: " + data.agencyName; var regBody = "New Agent Details:\n\n" + "Agent ID: " + agentId + "\n" + "Name: " + data.fullName + "\n" + "Agency: " + data.agencyName + "\n" + "Mobile: " + data.phone + "\n" + "City: " + data.city + "\n" + "Email: " + data.email; MailApp.sendEmail(alertEmail, regSubject, regBody); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Registration successful! Welcome to Journey Assist B2B Partner Network.", agent: { agentId: agentId, name: data.fullName, agency: data.agencyName, email: data.email, phone: data.phone, city: data.city } })).setMimeType(ContentService.MimeType.JSON); } // ------------------------------------------------------------- // 3. ACTION: AGENT SIGN IN // ------------------------------------------------------------- if (action === "signin") { var agentSheet = ss.getSheetByName("Agents"); if (!agentSheet) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: "No registered agents found. Please register first." })).setMimeType(ContentService.MimeType.JSON); } var agentData = agentSheet.getDataRange().getValues(); for (var i = 1; i < agentData.length; i++) { if (agentData[i][6] === data.email && agentData[i][7] === data.password) { return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Sign in successful!", agent: { agentId: agentData[i][1], name: agentData[i][2], agency: agentData[i][3], phone: agentData[i][4], city: agentData[i][5], email: agentData[i][6] } })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false, message: "Invalid email or password. Please try again." })).setMimeType(ContentService.MimeType.JSON); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } // GET API Function to fetch packages function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var packageSheet = ss.getSheetByName("Packages"); if (!packageSheet) { return ContentService.createTextOutput(JSON.stringify({ success: false, packages: [] })).setMimeType(ContentService.MimeType.JSON); } var data = packageSheet.getDataRange().getValues(); var packages = []; for (var i = 1; i < data.length; i++) { if (data[i][11] === "Active") { // Column L is status packages.push({ code: data[i][0], title: data[i][1], destination: data[i][2], duration: data[i][3], nightBreakup: data[i][4], category: data[i][5], b2bPrice: data[i][6], mrpPrice: data[i][7], inclusions: data[i][8], imageUrl: data[i][9], pdfLink: data[i][10] }); } } return ContentService.createTextOutput(JSON.stringify({ success: true, packages: packages })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })).setMimeType(ContentService.MimeType.JSON); } }
